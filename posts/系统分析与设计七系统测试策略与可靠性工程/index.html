一、测试金字塔——多层防御体系

1.1 经典测试金字塔

            ╱  E2E  ╲          ← 少量端到端测试（慢、脆弱、昂贵）
          ╱ 集成测试  ╲        ← 中量集成测试（服务间交互）
        ╱   单元测试    ╲      ← 大量单元测试（快、稳定、便宜）
      ╱____________________╲
层级占比速度稳定性覆盖内容
单元测试70%毫秒级⭐⭐⭐⭐⭐方法级逻辑、边界条件
集成测试20%秒级⭐⭐⭐⭐数据库、缓存、MQ集成
E2E测试10%分钟级⭐⭐完整业务流程

面试高频：为什么单元测试要多于E2E？

E2E测试运行慢、维护成本高、定位问题困难。单元测试快速反馈，能在开发阶段就发现问题。金字塔比例遵循"越底层回报越高"原则。

1.2 分布式系统的测试金字塔扩展

        ╱       混沌测试       ╲       ← 生产环境故障注入
      ╱        全链路压测        ╲     ← 容量评估与瓶颈发现
    ╱          契约测试            ╲   ← 服务间接口兼容性
  ╱           组件测试              ╲ ← 单服务独立验证
╱____________________________________╲

二、单元测试实战——从入门到精通

2.1 秒杀服务的单元测试（Java + JUnit5 + Mockito）

@ExtendWith(MockitoExtension.class)
class SeckillServiceTest {
    
    @Mock
    private RedissonClient redissonClient;
    
    @Mock
    private RocketMQTemplate rocketMQTemplate;
    
    @Mock
    private RScript rScript;
    
    @InjectMocks
    private SeckillService seckillService;
    
    /**
     * 核心场景1：正常秒杀成功
     */
    @Test
    void shouldSeckillSuccessfully() {
        // Given——准备数据
        Long userId = 1001L;
        Long goodsId = 2001L;
        String recordKey = "seckill:record:" + goodsId + ":" + userId;
        String stockKey = "seckill:stock:" + goodsId;
        
        // 幂等检查——未参与过
        RBucket<String> bucket = mock(RBucket.class);
        when(bucket.isExists()).thenReturn(false);
        when(redissonClient.getBucket(recordKey)).thenReturn(bucket);
        
        // 库存充足
        when(redissonClient.getScript()).thenReturn(rScript);
        when(rScript.eval(any(), anyString(), any(), anyList()))
            .thenReturn(1L); // 扣减成功
        
        // When——执行
        SeckillResult result = seckillService.executeSeckill(userId, goodsId);
        
        // Then——验证
        assertEquals(SeckillResult.Code.SUCCESS, result.getCode());
        verify(bucket).set(eq("1"), eq(30L), eq(TimeUnit.MINUTES));
        verify(rocketMQTemplate).asyncSend(eq("seckill-order-topic"), any(), any());
    }
    
    /**
     * 核心场景2：库存不足
     */
    @Test
    void shouldReturnSoldOutWhenNoStock() {
        Long userId = 1001L;
        Long goodsId = 2001L;
        
        RBucket<String> bucket = mock(RBucket.class);
        when(bucket.isExists()).thenReturn(false);
        when(redissonClient.getBucket(anyString())).thenReturn(bucket);
        
        when(redissonClient.getScript()).thenReturn(rScript);
        when(rScript.eval(any(), anyString(), any(), anyList()))
            .thenReturn(0L); // 库存不足
        
        SeckillResult result = seckillService.executeSeckill(userId, goodsId);
        
        assertEquals(SeckillResult.Code.SOLD_OUT, result.getCode());
        verify(rocketMQTemplate, never()).asyncSend(anyString(), any(), any());
    }
    
    /**
     * 核心场景3：重复秒杀（幂等性）
     */
    @Test
    void shouldRejectDuplicateSeckill() {
        Long userId = 1001L;
        Long goodsId = 2001L;
        
        RBucket<String> bucket = mock(RBucket.class);
        when(bucket.isExists()).thenReturn(true); // 已参与
        when(redissonClient.getBucket(anyString())).thenReturn(bucket);
        
        SeckillResult result = seckillService.executeSeckill(userId, goodsId);
        
        assertEquals(SeckillResult.Code.REPEAT, result.getCode());
        verify(rScript, never()).eval(any(), anyString(), any(), anyList());
    }
    
    /**
     * 核心场景4：MQ发送失败回滚库存
     */
    @Test
    void shouldRollbackWhenMqFails() {
        Long userId = 1001L;
        Long goodsId = 2001L;
        
        RBucket<String> bucket = mock(RBucket.class);
        when(bucket.isExists()).thenReturn(false);
        when(redissonClient.getBucket(anyString())).thenReturn(bucket);
        
        when(redissonClient.getScript()).thenReturn(rScript);
        when(rScript.eval(any(), anyString(), any(), anyList()))
            .thenReturn(1L);
        
        // 模拟MQ发送失败
        doAnswer(invocation -> {
            SendCallback callback = invocation.getArgument(2);
            callback.onException(new RuntimeException("MQ down"));
            return null;
        }).when(rocketMQTemplate).asyncSend(anyString(), any(), any());
        
        // 模拟原子计数器用于回滚
        RAtomicLong atomicLong = mock(RAtomicLong.class);
        when(redissonClient.getAtomicLong(anyString())).thenReturn(atomicLong);
        
        SeckillResult result = seckillService.executeSeckill(userId, goodsId);
        
        // 应该回滚库存
        verify(atomicLong).incrementAndGet();
        verify(bucket).delete();
    }
}

2.2 Go语言单元测试

package seckill

import (
    "context"
    "testing"
    "github.com/go-redis/redismock/v9"
    "github.com/stretchr/testify/assert"
)

func TestExecuteSeckill_Success(t *testing.T) {
    // Given
    ctx := context.Background()
    db, mock := redismock.NewClientMock()
    
    // Mock Exists: 未参与过
    mock.ExpectExists("seckill:record:2001:1001").SetVal(0)
    
    // Mock Lua: 库存扣减成功
    mock.ExpectEval(stockLuaScript, []string{"seckill:stock:2001"}).SetVal(1)
    
    // Mock SetEX: 标记已参与
    mock.ExpectSetEX("seckill:record:2001:1001", "1", 30*time.Minute).SetVal("OK")
    
    service := &SeckillService{rdb: db, producer: mockProducer}
    
    // When
    result, err := service.ExecuteSeckill(ctx, 1001, 2001)
    
    // Then
    assert.NoError(t, err)
    assert.Equal(t, 0, result.Code)
    assert.Equal(t, "秒杀成功，订单处理中", result.Msg)
}

func TestExecuteSeckill_SoldOut(t *testing.T) {
    ctx := context.Background()
    db, mock := redismock.NewClientMock()
    
    mock.ExpectExists("seckill:record:2001:1001").SetVal(0)
    mock.ExpectEval(stockLuaScript, []string{"seckill:stock:2001"}).SetVal(0) // 库存不足
    
    service := &SeckillService{rdb: db}
    
    result, err := service.ExecuteSeckill(ctx, 1001, 2001)
    
    assert.NoError(t, err)
    assert.Equal(t, 4, result.Code)
}

func TestExecuteSeckill_Duplicate(t *testing.T) {
    ctx := context.Background()
    db, mock := redismock.NewClientMock()
    
    mock.ExpectExists("seckill:record:2001:1001").SetVal(1) // 已参与
    
    service := &SeckillService{rdb: db}
    
    result, err := service.ExecuteSeckill(ctx, 1001, 2001)
    
    assert.NoError(t, err)
    assert.Equal(t, 2, result.Code)
}

四、全链路压测——发现系统瓶颈

4.1 压测方案设计

阶段目标持续时间QPS梯度
基准测试找到单机QPS上限5min1000→5000→10000
负载测试找到系统能力拐点15min逐级递增到失效
压力测试找到系统崩溃点10min持续增加直到崩溃
稳定性测试验证长时间稳定性2h50%峰值持续
尖峰测试验证突发能力3min瞬间从1000→100000

4.2 JMeter压测脚本设计

线程组配置：
- 线程数：1000（模拟1000并发用户）
- Ramp-Up：30秒（30秒内启动完毕）
- 循环：100次
- 持续时间：300秒

HTTP请求配置：
- 接口：/seckill/execute
- 参数化：userId, goodsId 从CSV读取
- 断言：响应码200 + body包含"success"
- 监听器：聚合报告 + 响应时间图 + 事务控制器

压测环境隔离：
- 独立压测集群（避免影响生产）
- 影子库：从生产复制数据，写入影子表
- 流量标识：压测流量染色标记

五、混沌工程——主动注入故障

5.1 混沌工程原则

🎯 核心思想：在生产环境主动注入故障，验证系统在真实故障下的表现，发现隐藏的缺陷。不是在找bug，是在验证系统的韧性。

混沌工程四步骤：

  1. 定义稳态（正常状态是什么？）
  2. 假设稳态会持续（注入故障后稳态会持续吗？）
  3. 注入真实事件（模拟服务器宕机、硬盘故障、网络延迟…）
  4. 证伪假设（如果稳态被打破，说明有隐藏问题）

5.2 ChaosBlade故障注入实战

# 1. CPU满载——验证限流是否生效

六、SRE可靠性工程

6.1 可靠性指标体系

SLI / SLO / SLA 三兄弟：

SLA（服务等级协议）——对外承诺
  ↑
SLO（服务等级目标）——对内目标（比SLA更严格）
  ↑
SLI（服务等级指标）——实际测量值

核心SLI指标：

类别SLI指标计算方式典型SLO
可用性请求成功率成功请求 / 总请求99.99%
延迟P99延迟99分位响应时间< 500ms
吞吐量QPS每秒请求数> 10000
错误率错误占比错误请求 / 总请求< 0.01%
饱和度资源使用率CPU/内存/磁盘使用率< 70%
持久性数据持久率未丢失数据比例99.9999%

6.2 错误预算——SRE的核心概念

错误预算 = 1 - SLO。例如SLO=99.99%，则每月错误预算=0.01%×30×24×60 = 4.32分钟。

错误预算策略：

  • 错误预算 > 0：可以发布新版本
  • 错误预算耗尽：冻结发布，全部资源投入稳定性
  • 错误预算消耗80%：触发告警

6.3 故障级别与响应

级别定义响应时间升级条件
P0核心功能完全不可用5min响应，30min恢复自动升级到CTO
P1核心功能部分不可用15min响应，2h恢复30min未恢复升级
P2非核心功能异常1h响应，24h恢复4h未恢复升级
P3轻微问题工作日处理—

7.2 Metrics——Prometheus + Grafana

黄金指标（RED + USE 方法）：

// Micrometer指标埋点
@Component
public class SeckillMetrics {
    
    private final MeterRegistry registry;
    
    // Rate：每秒请求数
    private final Counter requestCounter;
    
    // Errors：错误率
    private final Counter errorCounter;
    
    // Duration：延迟分布
    private final Timer seckillTimer;
    
    public SeckillMetrics(MeterRegistry registry) {
        this.registry = registry;
        
        this.requestCounter = Counter.builder("seckill.requests.total")
            .description("Total seckill requests")
            .tag("service", "seckill-service")
            .register(registry);
        
        this.errorCounter = Counter.builder("seckill.errors.total")
            .tag("service", "seckill-service")
            .register(registry);
        
        this.seckillTimer = Timer.builder("seckill.duration")
            .publishPercentiles(0.5, 0.95, 0.99) // P50, P95, P99
            .register(registry);
    }
    
    public void recordSuccess(long durationMs) {
        requestCounter.increment();
        seckillTimer.record(durationMs, TimeUnit.MILLISECONDS);
    }
    
    public void recordError(Throwable e) {
        requestCounter.increment();
        errorCounter.increment();
    }
}

7.3 Tracing——分布式链路追踪

OpenTelemetry + Jaeger集成：

@Configuration
public class TracingConfig {
    
    @Bean
    public OpenTelemetry openTelemetry() {
        // Jaeger Exporter
        JaegerGrpcSpanExporter exporter = JaegerGrpcSpanExporter.builder()
            .setEndpoint("http://jaeger-collector:14250")
            .build();
        
        // 采样策略：生产环境0.1%，测试环境100%
        Sampler sampler = Sampler.parentBased(
            Sampler.traceIdRatioBased(0.001));
        
        SdkTracerProvider tracerProvider = SdkTracerProvider.builder()
            .addSpanProcessor(BatchSpanProcessor.builder(exporter).build())
            .setSampler(sampler)
            .build();
        
        return OpenTelemetrySdk.builder()
            .setTracerProvider(tracerProvider)
            .buildAndRegisterGlobal();
    }
}

@Slf4j

            SeckillResult result = seckillService.executeSeckill(userId, goodsId);

        } finally {
            MDC.clear();
        }
    }

八、面试高频——测试与可靠性

Q1：如何保证秒杀系统上线后不出问题？

A：四重保障：

  1. 测试保障：单元→集成→契约→全链路压测，逐层覆盖
  2. 监控保障：Prometheus告警 + Grafana大盘 + ELK日志
  3. 容错保障：限流、熔断、降级、重试
  4. 预案保障：故障预案、回滚方案、值班制度

Q2：混沌工程和传统测试有什么区别？

传统测试混沌工程
已知输入，验证预期输出注入未知故障，观察系统反应
在测试环境在生产环境
找bug验证韧性
可预测不可预测

Q3：SLO设为99.99%意味着什么？

全年允许宕机时间 = (1 - 0.9999) × 365 × 24 × 60 ≈ 52.56分钟。 这意味着：每次部署有风险，错误预算有限，需要在"发布速度"和"稳定性"之间做权衡。

Q4：监控和可观测性的区别？

监控（Monitoring）：已知的未知——我知道可能出什么问题，我设置了告警 可观测性（Observability）：未知的未知——我不知道会出什么问题，但系统给我足够的信息让我快速定位

九、知识总结

核心知识矩阵

层次关键实践核心工具黄金法则
单元测试边界条件+异常路径JUnit5/Mockito/go test无单测不上线
集成测试真实中间件Testcontainers隔离外部依赖
契约测试提供方-消费方Pact接口变更需双方验证
全链路压测发现系统瓶颈JMeter/Locust压测环境=生产环境
混沌工程主动注入故障ChaosBlade不在测试环境注入
SRESLI/SLO/错误预算Prometheus/Grafana用数据驱动决策
可观测性日志+指标+链路ELK/Prometheus/Jaeger先建可观测再上线

可靠性工程口诀：

  1. 测（全量测试）：单元→集成→压测，层层把关
  2. 限（限流降级）：保护系统不被突发流量击垮
  3. 观（可观测性）：Logging/Metrics/Tracing三管齐下
  4. 混（混沌工程）：主动注入故障，验证系统韧性
  5. 预（应急预案）：有预案，有演练，有值班
